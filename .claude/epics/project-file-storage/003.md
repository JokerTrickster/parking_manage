interface VersionedFileUploaderProps {
  projectId: string;
  category: 'map' | 'cad' | 'roi';
  acceptedFormats?: string[];
}

const VersionedFileUploader: React.FC<VersionedFileUploaderProps> = ({
  projectId,
  category,
  acceptedFormats = ['*']
}) => {
  const { upload, uploading, progress, error } = useFileUpload();
  const { files, loading, refresh } = useFileList(projectId, category);

  const handleUpload = async (files: FileList) => {
    await upload(projectId, category, files);
    refresh(); // Reload file list
  };

  return (
    <Box>
      <Typography variant="h6">Upload {category.toUpperCase()} File</Typography>

      <Button variant="contained" component="label">
        Choose File
        <input
          type="file"
          hidden
          accept={acceptedFormats.join(',')}
          onChange={(e) => e.target.files && handleUpload(e.target.files)}
        />
      </Button>

      {uploading && <UploadProgress progress={progress} />}
      {error && <Alert severity="error">{error}</Alert>}

      <Typography variant="h6" sx={{ mt: 3 }}>Version History</Typography>
      <List>
        {files.map(file => (
          <ListItem key={file.filename}>
            <ListItemText
              primary={file.original_name}
              secondary={`Version: ${file.version} | Size: ${formatBytes(file.size_bytes)} | ${formatDate(file.upload_date)}`}
            />
            <IconButton onClick={() => downloadFile(projectId, category, file.filename)}>
              <DownloadIcon />
            </IconButton>
          </ListItem>
        ))}
      </List>
    </Box>
  );
};

const ImageFolderUploader: React.FC<ImageFolderUploaderProps> = ({
  projectId,
  category
}) => {

          hidden

### Custom Hooks

**useFileUpload Hook** (with chunking):
```typescript
const CHUNK_SIZE = 10 * 1024 * 1024; // 10MB

const useFileUpload = () => {
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [error, setError] = useState<string | null>(null);

  const upload = async (
    projectId: string,
    category: string,
    files: FileList
  ) => {
    setUploading(true);
    setProgress(0);
    setError(null);

    try {
      const formData = new FormData();

      // Check if files need chunking
      const totalSize = Array.from(files).reduce((sum, f) => sum + f.size, 0);

      if (totalSize > CHUNK_SIZE) {
        // Implement chunked upload
        await uploadInChunks(projectId, category, files, setProgress);
      } else {
        // Simple upload for small files
        Array.from(files).forEach(file => formData.append('files', file));

        const response = await axios.post(
          `/api/filestorage/${projectId}/${category}/upload`,
          formData,
          {
            headers: { 'Content-Type': 'multipart/form-data' },
            onUploadProgress: (progressEvent) => {
              const percent = (progressEvent.loaded / progressEvent.total) * 100;
              setProgress(percent);
            }
          }
        );

        if (!response.data.success) {
          throw new Error(response.data.message);
        }
      }
    } catch (err) {
      setError(err.message || 'Upload failed');
      throw err;
    } finally {
      setUploading(false);
    }
  };

  return { upload, uploading, progress, error };
};
```

**useFileList Hook**:
```typescript
const useFileList = (projectId: string, category: string) => {
  const [files, setFiles] = useState<FileMetadata[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchFiles = async () => {
    setLoading(true);
    setError(null);

    try {
      const response = await axios.get(
        `/api/filestorage/${projectId}/${category}/list`
      );

      if (response.data.success) {
        setFiles(response.data.data.files);
      } else {
        throw new Error(response.data.message);
      }
    } catch (err) {
      setError(err.message || 'Failed to fetch files');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchFiles();
  }, [projectId, category]);

  return { files, loading, error, refresh: fetchFiles };
};
```

**useFileDownload Hook**:
```typescript
const useFileDownload = () => {
  const download = async (
    projectId: string,
    category: string,
    filename: string
  ) => {
    try {
      const response = await axios.get(
        `/api/filestorage/${projectId}/${category}/download/${filename}`,
        { responseType: 'blob' }
      );

      // Create download link
      const url = window.URL.createObjectURL(new Blob([response.data]));
      const link = document.createElement('a');
      link.href = url;
      link.setAttribute('download', filename);
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.URL.revokeObjectURL(url);
    } catch (err) {
      console.error('Download failed:', err);
      throw err;
    }
  };

  return { download };
};
```
